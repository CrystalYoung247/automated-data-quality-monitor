SET MissingOwnerCount TO 0
Variables.CreateNewList List=> MissingOwnerIDs
SET MissingDepartmentCount TO 0
Variables.CreateNewList List=> MissingDepartmentIDs
LOOP FOREACH CurrentItem IN ExcelData
    IF IsEmpty(CurrentItem[2]) THEN
        Variables.IncreaseVariable Value: MissingOwnerCount IncrementValue: 1
        Variables.AddItemToList Item: CurrentItem[0] List: MissingOwnerIDs
    END
    IF IsEmpty(CurrentItem[1]) THEN
        Variables.IncreaseVariable Value: MissingDepartmentCount IncrementValue: 1
        Variables.AddItemToList Item: CurrentItem[0] List: MissingDepartmentIDs
    END
END
SET DataQualityReport TO $'''AUTOMATED DATA QUALITY REPORT

Missing Assigned Owners: %MissingOwnerCount%
Missing Departments: %MissingDepartmentCount%

Records Missing Assigned Owners:
%MissingOwnerIDs%

Records Missing Departments:
%MissingDepartmentIDs%'''
